  hosts: all
  become: yes
  tasks:
    - name: Install ngnix
      state: present
   - name: copy custom ngnix.conf
     copy: 
       src: ngnix.conf
       dest: /etc/ngnix/ngnix.conf
   - name: Start ngnix service
     service:
        name: ngnix
        enabled: yes
