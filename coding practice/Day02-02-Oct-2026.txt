- name: Install and start Ngnix on Pathnex
  hosts: all
  become: yes

  tasks:
    - name: Install ngnix
      yum:
        name: ngnix
        state: present

     - name: Enable ngnix
       service:
         name: ngnix
         state: started
         enabled: yes

}

  tags = {
    Name        = "Pathnex-Server"

  }
}

  name: pathnex-deployment

  replicas: 2

    matchLabels:

      labels:
        app: pathnex-app
     spec:
