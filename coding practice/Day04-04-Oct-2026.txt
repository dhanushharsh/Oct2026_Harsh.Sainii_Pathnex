- name: Install and Configure Ngnix
  hosts: all
  become: yes
  tasks:
   - name: Install ngnix
     yum: 
       name: ngnix
       state: present
   - name: Configure ngnix
     template: 
        src: ngnix.conf.j2
        dest: /etc/ngnix/ngnix.conf
   - name: Start ngnix service
     service: 
       name: ngnix
       state: started
       enabled: yes
