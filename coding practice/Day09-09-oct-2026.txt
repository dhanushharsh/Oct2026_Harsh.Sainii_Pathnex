  hosts:all
  become:yes
  tasks:
     - name: Pull Ngnix image
       docker_image:
          name: ngnix
          source: pull
     - name: Run Ngnix container 
       docker_container:
         name: ngnix-container
         image: ngnix
         state: started
         published_ports:
           - "8080:80"
